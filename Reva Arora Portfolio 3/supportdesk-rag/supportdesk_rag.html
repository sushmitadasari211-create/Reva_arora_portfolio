🛡️ How We Prevent Hallucinations

Four hard guarantees enforced at generation time — not best-effort prompting.

🔒
Context-Only Generation
LLM can only generate answers from retrieved context, not training data.
📎
Citation Requirement
Every factual claim must have a source citation attached.
🎚️
Confidence Threshold
Low-confidence answers (<60%) are flagged as "Uncertain".
🧑‍💻
Fallback to Human
If retrieval precision < 0.5, the query routes to human support.
# --- Multi-index setup: 5 LlamaIndex retrieval strategies ---
from llama_index.core import VectorStoreIndex, KeywordTableIndex, TreeIndex, SummaryIndex
from llama_index.core import StorageContext, Document
from llama_index.core.retrievers import QueryFusionRetriever

def build_indices(documents: list[Document]):
    """Build 5 complementary indices over the same ticket/doc corpus."""
    storage_context = StorageContext.from_defaults()

    vector_index  = VectorStoreIndex.from_documents(documents, storage_context=storage_context)
    keyword_index = KeywordTableIndex.from_documents(documents, storage_context=storage_context)
    tree_index    = TreeIndex.from_documents(documents, storage_context=storage_context)
    summary_index = SummaryIndex.from_documents(documents, storage_context=storage_context)

    # Hybrid index = fusion retriever combining vector + keyword via
    # Reciprocal Rank Fusion (RRF) -- see reranking section below.
    hybrid_retriever = QueryFusionRetriever(
        retrievers=[vector_index.as_retriever(similarity_top_k=8),
                    keyword_index.as_retriever()],
        mode="reciprocal_rerank",
        num_queries=1,  # disable query expansion for determinism
    )

    return {
        "vector": vector_index,
        "keyword": keyword_index,
        "hybrid": hybrid_retriever,
        "tree": tree_index,
        "summary": summary_index,
    }


# --- Anti-hallucination guard applied post-generation ---
def anti_hallucination_guard(answer: str, retrieved_chunks: list[str]) -> dict:
    """
    Rejects or flags any answer whose claims cannot be traced back to
    the retrieved context. This runs BEFORE the answer reaches the user.
    """
    context_blob = " ".join(retrieved_chunks).lower()
    answer_sentences = [s.strip() for s in answer.split(".") if s.strip()]

    unsupported = [
        s for s in answer_sentences
        if not any(keyword in context_blob for keyword in extract_keywords(s))
    ]

    grounded_ratio = 1 - (len(unsupported) / max(len(answer_sentences), 1))

    return {
        "grounded_ratio": round(grounded_ratio, 2),
        "unsupported_claims": unsupported,
        "passed": grounded_ratio >= 0.85,   # hard threshold, not a suggestion
    }


def extract_keywords(sentence: str) -> list[str]:
    stopwords = {"the", "a", "an", "is", "to", "of", "and", "in", "on", "for"}
    return [w for w in sentence.lower().split() if w not in stopwords and len(w) > 2]


# --- Two-layer evaluation harness ---
def evaluate_retrieval(index_name: str, retrieved_ids: list[str], gold_ids: list[str]) -> dict:
    """Layer 1: did we retrieve the RIGHT chunks at all?"""
    hits = [rid for rid in retrieved_ids if rid in gold_ids]
    precision_at_k = len(hits) / max(len(retrieved_ids), 1)

    rr = 0.0
    for rank, rid in enumerate(retrieved_ids, start=1):
        if rid in gold_ids:
            rr = 1 / rank
            break

    return {"index": index_name, "precision_at_k": round(precision_at_k, 2), "reciprocal_rank": round(rr, 2)}


def evaluate_generation(answer: str, guard_result: dict, has_citation: bool) -> dict:
    """Layer 2: given good retrieval, was the ANSWER actually good?"""
    return {
        "faithfulness": guard_result["grounded_ratio"],
        "hallucination_rate": round(1 - guard_result["grounded_ratio"], 2),
        "citation_present": has_citation,
        "confidence": round(guard_result["grounded_ratio"] * (1.0 if has_citation else 0.7), 2),
    }

from langchain_core.runnables import RunnablePassthrough, RunnableParallel, RunnableLambda
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser
from langchain.memory import ConversationBufferMemory

# --- Strict, no-hallucination system prompt ---
SYSTEM_PROMPT = """You are an IT support assistant. Answer ONLY using the provided context.
If the context does not contain the answer, respond exactly with:
"I don't have enough information in the knowledge base to answer this confidently."
Every claim you make MUST be traceable to a source in the context.
Always end your answer with a "Source:" line citing the ticket/doc IDs used."""

prompt = ChatPromptTemplate.from_messages([
    ("system", SYSTEM_PROMPT),
    ("human", "Context:\n{context}\n\nChat history:\n{history}\n\nQuestion: {question}"),
])

# --- Retrieval chain: fan out to the hybrid retriever, then rerank ---
def format_docs(docs):
    return "\n\n".join(f"[{d.metadata['source_id']}] {d.page_content}" for d in docs)

retrieval_chain = (
    RunnableParallel(
        docs=lambda x: hybrid_retriever.retrieve(x["question"]),
        question=lambda x: x["question"],
        history=lambda x: x["history"],
    )
    | RunnableLambda(lambda x: {**x, "context": format_docs(x["docs"])})
)

# --- Generation chain ---
generation_chain = prompt | llm | StrOutputParser()

# --- Full LCEL pipeline: retrieve -> rerank -> generate -> verify ---
rag_chain = (
    retrieval_chain
    | RunnableParallel(
        answer=generation_chain,
        docs=lambda x: x["docs"],   # carry docs through for the guard + citations
      )
    | RunnableLambda(lambda x: {
        **x,
        "verification": anti_hallucination_guard(x["answer"], [d.page_content for d in x["docs"]]),
      })
)

# --- Multi-turn memory integration ---
memory = ConversationBufferMemory(memory_key="history", return_messages=False)

def ask(question: str) -> dict:
    history = memory.load_memory_variables({})["history"]
    result = rag_chain.invoke({"question": question, "history": history})
    memory.save_context({"input": question}, {"output": result["answer"]})
    return result

# --- RAGAS-style metrics: retrieval precision, answer relevance, faithfulness ---

@dataclass
class RagasResult:
    context_precision: float   # did we retrieve relevant chunks, ranked well?
    answer_relevance: float    # does the answer actually address the question?
    faithfulness: float        # is every claim grounded in the retrieved context?
    hallucination_rate: float  # 1 - faithfulness, tracked explicitly for alerting

def run_ragas_eval(question, contexts, answer, ground_truth) -> RagasResult:

    answer_relevance   = score_answer_relevance(question, answer)
    faithfulness        = score_faithfulness(answer, contexts)
    return RagasResult(
        context_precision=context_precision,
        answer_relevance=answer_relevance,
        faithfulness=faithfulness,
        hallucination_rate=round(1 - faithfulness, 3),
    )

from langgraph.graph import StateGraph, END
from langgraph.checkpoint.memory import MemorySaver
from typing import TypedDict

class SupportState(TypedDict):
    question: str
    docs: list
    answer: str
    confidence: float
    needs_human: bool
    turn_count: int

# --- Graph nodes ---
def retrieve_node(state: SupportState) -> SupportState:
    docs = hybrid_retriever.retrieve(state["question"])
    return {**state, "docs": docs}

def generate_node(state: SupportState) -> SupportState:
    result = generation_chain.invoke({"context": format_docs(state["docs"]), "question": state["question"]})
    guard = anti_hallucination_guard(result, [d.page_content for d in state["docs"]])
    return {**state, "answer": result, "confidence": guard["grounded_ratio"]}

def confidence_gate(state: SupportState) -> str:
    """Route to human-in-the-loop if confidence is too low to answer safely."""
    if state["confidence"] < 0.6:
        return "human_handoff"
    return "respond"

def human_handoff_node(state: SupportState) -> SupportState:
    # This node raises an interrupt -- execution pauses here until a human
    # support agent reviews the query in the LangGraph Studio / custom UI.
    return {**state, "needs_human": True}

def respond_node(state: SupportState) -> SupportState:
    return {**state, "needs_human": False}


# --- Build the graph ---
graph = StateGraph(SupportState)
graph.add_node("retrieve", retrieve_node)
graph.add_node("generate", generate_node)
graph.add_node("human_handoff", human_handoff_node)
graph.add_node("respond", respond_node)

graph.set_entry_point("retrieve")
graph.add_edge("retrieve", "generate")
graph.add_conditional_edges("generate", confidence_gate, {
    "human_handoff": "human_handoff",
    "respond": "respond",
})
graph.add_edge("respond", END)

# Interrupt BEFORE human_handoff executes, so a real person confirms/edits
# the escalation rather than the agent silently deciding on its own.
checkpointer = MemorySaver()
app = graph.compile(checkpointer=checkpointer, interrupt_before=["human_handoff"])


# --- Multi-turn usage with persistent memory across a conversation thread ---
config = {"configurable": {"thread_id": "user-4821-session"}}

def chat_turn(question: str):
    result = app.invoke({"question": question, "turn_count": 1}, config=config)
    if result.get("needs_human"):
        print("⚠️  Low confidence -- escalated to a human support agent for review.")
    else:
        print(result["answer"])
    return result
