Turnover Cost Calculation

# Estimates the fully-loaded cost of a single departing employee
def turnover_cost_per_exit(salary, time_to_fill_days, ramp_months, criticality):
    replacement_cost   = salary * 0.20                                   # agency + advertising + admin
    recruiter_cost     = salary * 0.10                                   # internal recruiter time
    manager_time_cost  = (salary / 2080) * 15 * 2                        # hourly rate * interview hours * interviewers
    onboarding_cost    = salary * (ramp_months / 12) * 0.50              # training + ramp materials
    productivity_loss  = salary * (ramp_months / 12) * criticality       # ramp-up inefficiency + team drag
    vacancy_cost       = (salary / 260) * time_to_fill_days * 0.60       # lost output while seat is empty

    return {
        "replacement": replacement_cost,
        "recruiter": recruiter_cost,
        "manager_time": manager_time_cost,
        "onboarding": onboarding_cost,
        "productivity_loss": productivity_loss,
        "vacancy": vacancy_cost,
        "total": (replacement_cost + recruiter_cost + manager_time_cost
                  + onboarding_cost + productivity_loss + vacancy_cost),
    }

Segment Risk Scoring

# Scores a department x tenure-band segment on turnover cost concentration
def segment_risk_cost(dept_exits, tenure_share, tenure_risk_mult, cost_per_exit):
    # tenure_share: fraction of exits that fall inside this tenure band
    # tenure_risk_mult: early-tenure bands carry more onboarding/ramp risk
    return dept_exits * tenure_share * tenure_risk_mult * cost_per_exit


def attrition_risk_score(attrition_multiplier, tenure_risk_weighted, max_score=100):
    # Normalizes a department's relative flight risk onto a 0-100 scale
    raw_score = attrition_multiplier * tenure_risk_weighted
    return min(max_score, round(raw_score * 40))

Intervention ROI Formula

# Converts an investment level (0-100%) into cost, exits avoided, and ROI
def simulate_intervention(level_pct, employees_targeted, avg_salary,
                           cost_factor, max_attrition_reduction,
                           baseline_exits, avg_cost_per_exit):
    level = level_pct / 100

    implementation_cost = cost_factor * employees_targeted * avg_salary * level
    attrition_reduction = max_attrition_reduction * level          # diminishing returns can be layered in
    exits_avoided = baseline_exits * attrition_reduction
    savings = exits_avoided * avg_cost_per_exit

    net_impact = savings - implementation_cost
    roi_pct = (net_impact / implementation_cost * 100) if implementation_cost > 0 else 0

    return {
        "cost": implementation_cost,
        "exits_avoided": exits_avoided,
        "savings": savings,
        "net_impact": net_impact,
        "roi_pct": roi_pct,
    }
