    1. Breach Cost Estimation Formula
    def estimate_breach_cost(global_avg, industry_multiplier, size_multiplier,
                          region_multiplier, sensitivity_weight, maturity_multiplier):
    """
    Estimated breach cost = global baseline x every risk-profile factor.
    Each multiplier is >1 when it INCREASES cost, <1 when it REDUCES it.
    """
    base_cost = global_avg * industry_multiplier * size_multiplier * region_multiplier
    base_cost *= sensitivity_weight       # PII / financial data raises the stakes
    base_cost *= maturity_multiplier      # stronger security posture lowers the cost
    return base_cost

# Example: mid-size technology company, US region, medium sensitivity, 3-star maturity
cost = estimate_breach_cost(
    global_avg=4_880_000,
    industry_multiplier=1.05,
    size_multiplier=0.85,
    region_multiplier=2.09,
    sensitivity_weight=1.0,
    maturity_multiplier=0.75,
)
print(f"Estimated breach cost: ${cost:,.0f}")
    2. Detection Speed Cost Curve
    def days_cost_multiplier(days):
    """
    Cost escalates with dwell time. The curve is roughly flat before the
    200-day benchmark, then bends sharply upward past it (the "200-day cliff").
    Anchors: ~0.79x at 200 days ($3.87M / $4.88M), accelerating after.
    """
    if days <= 200:
        return 0.60 + (days / 200) * 0.193   # 0.60x at day 0 -> 0.793x at day 200
    else:
        overshoot = (days - 200) / 165        # normalized 0..1 across 200->365
        return 0.793 + (overshoot ** 1.3) * 0.6

def cost_at_days(profile_base_cost, days, maturity_multiplier):
    return profile_base_cost * days_cost_multiplier(days) * maturity_multiplier

for d in [90, 200, 247, 365]:
    print(d, "days ->", round(days_cost_multiplier(d), 3), "x multiplier")
    3. Security Investment ROI Ranking
    def rank_investments(investments, current_cost):
    """
    investments: list of dicts with 'name', 'implementation_cost', 'risk_reduction_pct'
    Ranks by ROI = (dollar_savings - implementation_cost) / implementation_cost
    """
    ranked = []
    for inv in investments:
        savings = current_cost * inv["risk_reduction_pct"]
        roi = (savings - inv["implementation_cost"]) / inv["implementation_cost"]
        payback_months = inv["implementation_cost"] / (savings / 12) if savings > 0 else float("inf")
        ranked.append({**inv, "savings": savings, "roi": roi, "payback_months": payback_months})
    return sorted(ranked, key=lambda x: x["roi"], reverse=True)

best = rank_investments(investments, current_cost)[0]
print("Best ROI pick:", best["name"], f"ROI={best['roi']:.1%}")
